// Google Apps Script - Code.gs for Al-Rowad Cover Hub & Observation Tracker

function doGet(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheets = ss.getSheets();
    var teachers = [];
    var timetable = [];
    var coverLogs = [];
    var sectionHeads = [];
    var observations = [];
    var DAYS = ["Sunday", "Monday", "Tuesday", "Wednesday", "Thursday"];

    var logSheet = ss.getSheetByName("CoverAssignments");
    if (logSheet && logSheet.getLastRow() > 1) {
      var logData = logSheet.getRange(2, 1, logSheet.getLastRow() - 1, 14).getValues();
      logData.forEach(function(row) {
        if (row[0]) {
          coverLogs.push({
            id: String(row[0]),
            date: String(row[1]),
            day: String(row[2]),
            period: String(row[3]),
            absentTeacherId: String(row[4] || ""),
            absentTeacherName: String(row[5] || ""),
            coverTeacherId: String(row[6] || ""),
            coverTeacherName: String(row[7] || ""),
            subject: String(row[8] || ""),
            coverType: String(row[9] || ""),
            authorizer: String(row[10] || ""),
            notes: String(row[11] || ""),
            overrideReason: String(row[12] || ""),
            timestamp: String(row[13] || "")
          });
        }
      });
    }

    var shSheet = ss.getSheetByName("SectionHeads");
    if (shSheet && shSheet.getLastRow() > 1) {
      var shData = shSheet.getRange(2, 1, shSheet.getLastRow() - 1, 6).getValues();
      shData.forEach(function(row) {
        if (row[0]) {
          sectionHeads.push({
            section: String(row[0]),
            name: String(row[1]),
            title: String(row[2]),
            email: String(row[3]),
            office: String(row[4]),
            contact: String(row[5])
          });
        }
      });
    }

    var obsSheet = ss.getSheetByName("Observations");
    if (obsSheet && obsSheet.getLastRow() > 1) {
      var obsData = obsSheet.getRange(2, 1, obsSheet.getLastRow() - 1, 8).getValues();
      obsData.forEach(function(row) {
        if (row[0]) {
          observations.push({
            id: String(row[0]),
            date: String(row[1]),
            day: String(row[2]),
            period: String(row[3]),
            teacherId: String(row[4]),
            teacherName: String(row[5]),
            status: String(row[6] || "Completed"),
            notes: String(row[7] || "")
          });
        }
      });
    }

    sheets.forEach(function(sheet) {
      var sName = sheet.getName();
      if (sName !== "CoverAssignments" && sName !== "SectionHeads" && sName !== "Observations" && sName !== "Sheet1" && sheet.getLastRow() >= 7) {
        var vals = sheet.getRange(1, 1, 7, 9).getValues();
        var teacherName = String(vals[0][1] || sName).trim();
        var teacherId = String(vals[0][3] || ("T_" + sName.replace(/[^a-zA-Z0-9]/g, ""))).trim();
        
        var teacherSec = String(vals[0][5] || "").trim().toUpperCase();
        if (!teacherSec || teacherSec === "LG/MG") {
          var altSec = String(vals[0][4] || "").trim().toUpperCase();
          if (["LG", "MG", "HG", "IG"].indexOf(altSec) !== -1) {
            teacherSec = altSec;
          }
        }
        if (!teacherSec || teacherSec === "LG/MG") teacherSec = "LG";

        var isHodVal = (String(vals[0][7] || "").toUpperCase() === "YES" || String(vals[0][7] || "").toUpperCase() === "TRUE");

        if (teacherName) {
          teachers.push({
            id: teacherId,
            name: teacherName,
            section: teacherSec,
            isHoD: isHodVal,
            department: isHodVal ? "Head of Department" : "Staff Teacher"
          });

          for (var r = 2; r < 7; r++) {
            var dayName = String(vals[r][0]).trim();
            if (DAYS.indexOf(dayName) !== -1) {
              for (var c = 1; c <= 8; c++) {
                var pName = "P" + c;
                var subjVal = String(vals[r][c] || "").trim();
                var isFree = (!subjVal || subjVal === "-" || subjVal.toUpperCase() === "FREE");
                timetable.push({
                  teacherId: teacherId,
                  teacherName: teacherName,
                  section: teacherSec,
                  isHoD: isHodVal,
                  day: dayName,
                  period: pName,
                  subject: isFree ? "FREE" : subjVal,
                  room: "N/A",
                  isFree: isFree
                });
              }
            }
          }
        }
      }
    });

    return ContentService.createTextOutput(JSON.stringify({
      status: "success",
      teachers: teachers,
      timetable: timetable,
      coverLogs: coverLogs,
      sectionHeads: sectionHeads,
      observations: observations,
      timestamp: new Date().toISOString()
    })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error",
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();

    if (data.action === "LOG_COVER") {
      var logSheet = ss.getSheetByName("CoverAssignments") || ss.insertSheet("CoverAssignments");
      if (logSheet.getLastRow() === 0) {
        logSheet.appendRow(["Log ID", "Date", "Day", "Period", "Absent Teacher ID", "Absent Teacher", "Cover Teacher ID", "Cover Teacher", "Subject", "Duty Type", "Authorizer", "Notes", "Override Reason", "Timestamp"]);
      }
      logSheet.appendRow([
        data.coverLog.id, data.coverLog.date, data.coverLog.day, data.coverLog.period,
        data.coverLog.absentTeacherId || "", data.coverLog.absentTeacherName,
        data.coverLog.coverTeacherId || "", data.coverLog.coverTeacherName,
        data.coverLog.subject, data.coverLog.coverType, data.coverLog.authorizer,
        data.coverLog.notes || "", data.coverLog.overrideReason || "", data.coverLog.timestamp
      ]);
    } else if (data.action === "SYNC_ALL" || data.timetable) {
      var teachers = data.teachers || [];
      var timetable = data.timetable || [];
      var sectionHeads = data.sectionHeads || [];
      var coverLogs = data.coverLogs || [];
      var observations = data.observations || [];
      var DAYS = ["Sunday", "Monday", "Tuesday", "Wednesday", "Thursday"];
      var PERIODS = ["P1", "P2", "P3", "P4", "P5", "P6", "P7", "P8"];

      if (coverLogs.length >= 0) {
        var logSheet = ss.getSheetByName("CoverAssignments") || ss.insertSheet("CoverAssignments");
        logSheet.clear();
        logSheet.appendRow(["Log ID", "Date", "Day", "Period", "Absent Teacher ID", "Absent Teacher", "Cover Teacher ID", "Cover Teacher", "Subject", "Duty Type", "Authorizer", "Notes", "Override Reason", "Timestamp"]);
        coverLogs.forEach(function(l) {
          logSheet.appendRow([
            l.id || "", l.date || "", l.day || "", l.period || "",
            l.absentTeacherId || "", l.absentTeacherName || "",
            l.coverTeacherId || "", l.coverTeacherName || "",
            l.subject || "", l.coverType || "", l.authorizer || "",
            l.notes || "", l.overrideReason || "", l.timestamp || ""
          ]);
        });
        var logHeader = logSheet.getRange(1, 1, 1, 14);
        logHeader.setBackground("#0369A1");
        logHeader.setFontColor("#FFFFFF");
        logHeader.setFontWeight("bold");
      }

      if (observations.length >= 0) {
        var obsSheet = ss.getSheetByName("Observations") || ss.insertSheet("Observations");
        obsSheet.clear();
        obsSheet.appendRow(["Observation ID", "Date", "Day", "Period", "Teacher ID", "Teacher Name", "Status", "Confidential Notes"]);
        observations.forEach(function(o) {
          obsSheet.appendRow([
            o.id || "", o.date || "", o.day || "", o.period || "",
            o.teacherId || "", o.teacherName || "", o.status || "Completed",
            o.notes || ""
          ]);
        });
        var obsHeader = obsSheet.getRange(1, 1, 1, 8);
        obsHeader.setBackground("#BE123C");
        obsHeader.setFontColor("#FFFFFF");
        obsHeader.setFontWeight("bold");
      }

      if (sectionHeads.length > 0) {
        var shSheet = ss.getSheetByName("SectionHeads") || ss.insertSheet("SectionHeads");
        shSheet.clear();
        shSheet.appendRow(["Section", "Name", "Title", "Email", "Office", "Contact"]);
        sectionHeads.forEach(function(sh) {
          shSheet.appendRow([sh.section || "", sh.name || "", sh.title || "", sh.email || "", sh.office || "", sh.contact || ""]);
        });
        var shHeader = shSheet.getRange(1, 1, 1, 6);
        shHeader.setBackground("#6B21A8");
        shHeader.setFontColor("#FFFFFF");
        shHeader.setFontWeight("bold");
      }

      teachers.forEach(function(t) {
        var sheetName = t.name.replace(/[\/\\?*:[\]]/g, "_").substring(0, 30);
        var teacherSheet = ss.getSheetByName(sheetName) || ss.insertSheet(sheetName);
        teacherSheet.clear();

        var row1 = ["Teacher Name", t.name, "ID", t.id || "", "Section", t.section || "LG", "HoD", t.isHoD ? "YES" : "NO", ""];
        var row2 = ["Day / Lesson", "P1", "P2", "P3", "P4", "P5", "P6", "P7", "P8"];
        var rows = [row1, row2];

        var teacherSlots = timetable.filter(function(s) { return s.teacherId === t.id; });

        DAYS.forEach(function(day) {
          var row = [day];
          PERIODS.forEach(function(p) {
            var slot = teacherSlots.find(function(s) { return s.day === day && s.period === p; });
            var val = (slot && !slot.isFree && slot.subject !== "FREE") ? slot.subject : "";
            row.push(val);
          });
          rows.push(row);
        });

        var range = teacherSheet.getRange(1, 1, rows.length, 9);
        range.setValues(rows);

        var headerRange = teacherSheet.getRange(1, 1, 2, 9);
        headerRange.setBackground("#FFFF00");
        headerRange.setFontWeight("bold");

        var fullGridRange = teacherSheet.getRange(1, 1, 7, 9);
        fullGridRange.setBorder(true, true, true, true, true, true, "#000000", SpreadsheetApp.BorderStyle.SOLID);
        fullGridRange.setHorizontalAlignment("center");
        fullGridRange.setVerticalAlignment("middle");

        teacherSheet.setColumnWidth(1, 130);
        for (var col = 2; col <= 9; col++) {
          teacherSheet.setColumnWidth(col, 85);
        }
      });
    }

    return ContentService.createTextOutput(JSON.stringify({ result: "success" })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ result: "error", message: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}
